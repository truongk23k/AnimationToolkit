// Template: a new animator. Keep Sample() a pure function of t so it loops and bakes.
using AnimationToolkit;
using UnityEngine;

[AddComponentMenu("Animation Toolkit/Example Wobble Animator")]
public class WobbleAnimator : OffsetAnimator
{
    [Tooltip("Rotation amplitude in degrees.")]
    [SerializeField] float degrees = 6f;
    [SerializeField, Min(0f)] float frequency = 0.8f;

    protected override void Sample(float t, float dt, ref AnimPose pose)
    {
        float s = Cycle(t, frequency, 0f);
        pose.rotation = Quaternion.Euler(0f, 0f, degrees * s);
        pose.scale = new Vector3(1f + 0.02f * s, 1f - 0.02f * s, 1f); // squash & stretch
    }

    public override void SnapToLoop(float duration) { frequency = SnapFrequency(frequency, duration); }
}
